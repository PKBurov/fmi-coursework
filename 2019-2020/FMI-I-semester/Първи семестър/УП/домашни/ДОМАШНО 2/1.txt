double distance(double x1, double y1, double x2, double y2)
{


	double a = (x2-x1)*(x2-x1);
	double b = (y2-y1)*(y2-y1);

	return sqrt(a+b);
}

}

int quadrant(double a, double b)
{
	int first = 1, second = 2, third = 3, fourth = 4;
	if (a > 0 && b > 0)
		return first;

	if (a < 0 && b>0)
		return second;

	if (a < 0 && b < 0)
		return third;

	if (a > 0 && b < 0)
		return fourth;

	if (a == 0 || b == 0)
		return 0;
}
