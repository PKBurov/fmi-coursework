#include<iostream>
#include<cmath>
#include <iomanip>  

using namespace std;
int main()
{
	

	int n, t, v;
	double d;
	double vN,arr,sN,lastvN=0;

	double s = 0;
	cin >> n >> d;
	for (int i = 0; i < n; i++)
	{
		cin >> arr;
		vN = lastvN + arr*d;
		if (vN < 0)
			vN = 0;
		sN = lastvN*d + 0.5*arr*d*d;
		if (sN > 0)
			s += sN;
		lastvN = vN;
	}

	cout << setprecision(5) << s << '\n';
	return 0;
}
