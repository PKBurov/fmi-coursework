
#include<iostream>  
using namespace std;


void selectionSort(unsigned int fnumber[], double tasks[],  double points[], int n)
{
	
	int minElement, minElementTask, minElementPoint;
	int changeFN;
	double changeTasks,changePoints;

	for (int i = 0; i < n - 1; i++)
	{
		minElement = i;

		for (int j = i + 1; j < n; j++)
		{

			if (fnumber[j] < fnumber[minElement])
				minElement = j;
			
		}

		if (minElement != i)
		{
			changeFN = fnumber[i];
			fnumber[i] = fnumber[minElement];
			fnumber[minElement] = changeFN;

			
			changeTasks = tasks[i];
			tasks[i] = tasks[minElement];
			tasks[minElement] = changeTasks;

			changePoints = points[i];
			points[i] = points[minElement];
			points[minElement] = changePoints;


		}
	}

	for (int j = 0; j < n-1; j++)
	{
		if (fnumber[j] == fnumber[j + 1] && tasks[j] > tasks[j + 1])
		{	
			swap(tasks[j], tasks[j + 1]);
			swap(points[j], points[j + 1]);
		}

		if (fnumber[j] == fnumber[j + 1] && tasks[j] == tasks[j + 1] && points[j] > points[j + 1])
		{
			swap(tasks[j], tasks[j + 1]);
			swap(points[j], points[j + 1]);
		}
	}


	for (int t = 0; t < n; t++)
	{
		cout << fnumber[t] << " "
			<< tasks[t] << " "
			<< points[t] << '\n';

	}
}

int main()
{
	unsigned int Fnumbers[100];
	double tasks[100];
	double points[100];

	unsigned int n;
	cin >> n;

	for (int i = 0; i < n; i++)
	{
		cin >> Fnumbers[i];
		cin >> tasks[i];
		cin >> points[i];

	}

	selectionSort(Fnumbers, tasks, points, n);
	return 0;
}
