{

    int matrixsA[ROWS][COLUMNS],matrixsB[ROWS][COLUMNS];

    for(int i=0;i<ROWS;i++)
    {
        for(int j=0;j<COLUMNS;j++)

            cout<<" matrixsA["<<i<<"]["<<j<<"]=";

            cout<<" matrixsB["<<i<<"]["<<j<<"]=";
        cin>>matrixsB[i][j];
        }

        for(int j=0;j<COLUMNS;j++)
        {
        SUM[i][j]=  matrixsA[i][j]+matrixsB[i][j];
        }


    }
    cout<<"\n";
