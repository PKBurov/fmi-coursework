int lastDigit(int n)
{
	return n % 10;
}

int cutLastDigit(int n)
{
	return n / 10;
}

int toBaseTen(int currentBase, int n)
{
	int nInBaseTen = 0, multiplier = 1;
	do
	{
		nInBaseTen = nInBaseTen + lastDigit(n) * multiplier;
		n = cutLastDigit(n);
		multiplier = multiplier * currentBase;
	} while (n > 0);
	return nInBaseTen;
}

int Base(char number[])
{
	int t = 0;
	if (number[t] == '0' && number[t + 1] != '\0' && number[t + 1] != 'b' && number[t + 1] != 'x')   return 8;
	if (number[t] == '0' && number[t + 1] == 'b')   return 2;
	if (number[t] == '0' && number[t + 1] == 'x')   return 16;

	if (number[t] != '0')   return 10;
	if (number[t] == '0' && number[t + 1] == '\0')   return 10;

}

char* substr(char* arr, int begin, int len)
{
	ConvertToNumber(arr, len);
	char* res = new char[len];
	for (int i = 0; i < len; i++)
		res[i] = *(arr + begin + i);
	res[len] = 0;
	return res;
}
